def read(p):
    return pd.read_csv(p, sep='\t', float_precision='round_trip')

def write(d, n):
    if not isinstance(d, pd.DataFrame): d = pd.DataFrame(d)
    d.to_csv(OUT/n, sep='\t', index=False, na_rep='', float_format='%.17g')

def sha(p):
    h=hashlib.sha256()
    with open(p,'rb') as f:
        for b in iter(lambda:f.read(8*1024*1024),b''): h.update(b)
    return h.hexdigest()

def load(p,n):
    spec=importlib.util.spec_from_file_location(n,p); m=importlib.util.module_from_spec(spec)
    sys.modules[n]=m; spec.loader.exec_module(m); return m

def check(n,ok,detail=''):
    CHECKS.append(dict(check=n,passed=bool(ok),detail=str(detail)))
    if not ok: raise AssertionError((n,detail))

def joined(v): return ';'.join(sorted(set(str(x) for x in v if pd.notna(x) and str(x))))
def strings(v): return set(str(v).split(';')) if pd.notna(v) and str(v) else set()

def extract(t):
    """Targeted streaming extraction preserving original position/min-P semantics.

    Adds allele provenance; gene score/count replay against the completed full
    GWAS stage25 replay is a mandatory gate. Never reconstruct P from beta/SE
    except the already-authorized AST VCF fallback in the original loader.
    """
    old,extra,score,_=resources();q=configuration(t,old,extra);path=Path(q['path'])
    original=json.loads((ST25/f'{t}_manifest.json').read_text());before=sha(path)
    check(t+'_raw_hash',before==original['input_sha256'])
    variants=read(OUT/'variant_dictionary.tsv.gz');targets=set(zip(variants.chromosome,variants.position_grch37)); selected={}; counts=Counter()
    opening=gzip.open if path.suffix=='.gz' else open
    with opening(path,'rt') as stream:
        header=None
        for line in stream:
            if q['format']=='opengwas_vcf':
                if line.startswith('#'):continue
                p=line.rstrip('\n').split('\t');counts['raw_rows']+=1
                if len(p)<10:continue
                ch=old.parse_chromosome(p[0])
                try: pos=int(p[1])
                except ValueError:continue
                if (ch,pos) not in targets:continue
                m=dict(zip(p[8].split(':'),p[9].split(':')))
                try:
                    value=max(10.**(-float(m['LP'])),1e-300) if m.get('LP') not in (None,'','.') else old.z_to_p(float(m['ES'])/float(m['SE']))
                except (ValueError,ZeroDivisionError):continue
                ea,oa=p[4].upper(),p[3].upper()
            else:
                if line.startswith('##'):continue
                if header is None:
                    header=line.strip().lstrip('#').split(q.get('sep')); ix={n:i for i,n in enumerate(header)};continue
                p=line.strip().split(q.get('sep'));counts['raw_rows']+=1
                try:
                    ch=old.parse_chromosome(p[ix[q['chr_col']]])
                    pos=int(float(p[ix[q['pos_col']]])) if t in extra.TRAITS else int(p[ix[q['pos_col']]])
                except (ValueError,IndexError):continue
                if (ch,pos) not in targets:continue
                try: value=float(p[ix[q['p_col']]])
                except (ValueError,IndexError):continue
                if not math.isfinite(value) or not 0<=value<=1:continue
                value=max(value,1e-300)
                ea=p[ix[q['ea_col']]].upper() if q['ea_col'] in ix else ''
                oa=p[ix[q['nea_col']]].upper() if q['nea_col'] in ix else ''
            if not math.isfinite(value) or not 0<=value<=1:continue
            counts['target_valid_rows']+=1;key=(ch,pos)
            pair='/'.join(sorted([ea,oa])) if ea and oa and ea!=oa and ea not in {'.','NA'} and oa not in {'.','NA'} and ',' not in ea and ',' not in oa else 'UNRESOLVED'
            if key not in selected:selected[key]=dict(p=value,pairs={pair},all_pairs={pair},n_records=1,conflicting_p=False)
            else:
                r=selected[key];r['n_records']+=1;r['all_pairs'].add(pair);r['conflicting_p'] |= value!=r['p']
                if value<r['p']:r['p']=value;r['pairs']={pair}
                elif value==r['p']:r['pairs'].add(pair)
    records=[]
    for _,v in variants.iterrows():
        r=selected.get((v.chromosome,v.position_grch37))
        if r is None:continue
        match=len(r['pairs'])==1 and v.allele_pair in r['pairs']
        records.append(dict(variant_index=int(v.variant_index),p=r['p'],selected_source_allele_pairs=joined(r['pairs']),all_source_allele_pairs=joined(r['all_pairs']),n_source_records=r['n_records'],coordinate_P_conflict=r['conflicting_p'],allele_verified=match,allele_status='exact_unordered_pair' if match else ('ambiguous_minimum_P_alleles' if len(r['pairs'])>1 else 'literal_pair_mismatch_or_missing')))
    inputs=pd.DataFrame(records);write(inputs,f'inputs_{t}.tsv.gz')
    edges=read(OUT/'frozen_candidate_mapping_edges.tsv.gz'); actual=edges.merge(inputs,on='variant_index',validate='many_to_one')
    expected=read(ST25/f'{t}_gene_acat.tsv').set_index('gene');rows=[]
    for name,x in actual.groupby('gene_symbol',sort=False):
        pv=x.p.to_numpy(); observed=score.boundary(pv);r=expected.loc[name]
        ok=len(x)==r.nsnps_matched and np.isclose(observed,r.boundary_p,rtol=1e-11,atol=0) and pv.min()==r.min_p
        rows.append(dict(gene_symbol=name,trait=t,n_records=len(x),source_n_records=r.nsnps_matched,score=observed,source_score=r.boundary_p,minimum_P=pv.min(),source_minimum_P=r.min_p,passed=ok))
    validation=pd.DataFrame(rows);write(validation,f'replay_{t}.tsv');check(t+'_all_gene_input_replays',validation.passed.all())
    candidate_genes=set(read(ST28/'candidate_universe.tsv').gene_symbol)
    check(t+'_evaluable_membership',set(validation.gene_symbol)==set(expected.loc[expected.index.isin(candidate_genes)&expected.boundary_p.notna()].index))
    check(t+'_raw_unchanged',before==sha(path))
    m=dict(trait=t,path=str(path.relative_to(ROOT)),sha256=before,counts=dict(counts),coordinate_keys=len(selected),actual_variant_keys=len(inputs),allele_status_counts=inputs.allele_status.value_counts().to_dict(),replay_genes=len(validation),all_replays_passed=True,script_sha256=sha(OUT/'run.py'))
    (OUT/f'extract_{t}.json').write_text(json.dumps(m,indent=2)+'\n');write(CHECKS,f'checks_{t}.tsv');print(json.dumps(m),flush=True)

def read_pair_keys(d):
    """Strict schema check; do not infer by column position."""
    result=[]
    if {'chr','pos','a1','a2'}<=set(d):
        for r in d.itertuples(index=False):
            c=int(r.chr);p=int(r.pos);a=sorted([str(r.a1).upper(),str(r.a2).upper()]);result.append((f'{c}:{p}',f'GRCh37:{c}:{p}:'+':'.join(a)))
    elif 'snp' in d or 'key' in d:
        for s in d['snp' if 'snp' in d else 'key']:
            z=str(s).split(':');assert len(z)==4, s;c=int(z[0].removeprefix('chr'));p=int(z[1]);a=sorted([z[2].upper(),z[3].upper()]);result.append((f'{c}:{p}',f'GRCh37:{c}:{p}:'+':'.join(a)))
    else:raise ValueError(('Unrecognized existing pair schema',list(d.columns)))
    return result
