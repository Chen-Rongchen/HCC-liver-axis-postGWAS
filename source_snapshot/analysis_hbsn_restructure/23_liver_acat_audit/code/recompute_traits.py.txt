import argparse
import json
import sys
from pathlib import Path
from datetime import datetime, timezone
import numpy as np
import pandas as pd

OUT=Path(__file__).resolve().parents[1]
ROOT=OUT.parents[1]
sys.path.insert(0,str(ROOT/'analysis_hbsn_restructure/22_acat_numerical_audit/code'))
from numerical_audit import load, sha, stable_acat
MAP=ROOT/'analysis_update_2026_hcc_gwas/03_qc/frozen_map_cache'
old=load(ROOT/'scripts/02_gene_level_liver_axis_nomination/02_rerun_acat_9trait.py','liver_original_adapter')
extra=load(ROOT/'scripts/17_trait_eligibility/17_audit_psc_aih_gene_inputs.py','liver_extra_adapter')
TRAITS=['CIRR','LIVER_FAT','NAFLD','ALT','AST','GGT','ALP','BILI','PBC','PSC','AIH']

def write(df,name):df.to_csv(OUT/name,sep='\t',index=False,na_rep='',float_format='%.17g')

def run(trait):
    start=datetime.now(timezone.utc).isoformat()
    if trait in extra.TRAITS:
        config=extra.TRAITS[trait]; path=Path(config['path']); before=sha(path)
        snps,raw=extra.stream_gwas(config)
        saved=ROOT/f'result/17_trait_eligibility/acat_{trait}_eligibility.tsv'
    else:
        config=old.configured_gwas('native_grch37_primary')[trait]; path=Path(config['path']); before=sha(path)
        snps=old.load_gwas_snp_map(config,trait)
        saved=ROOT/f'result/08_evidence_completion/acat_native_grch37_primary/acat_{trait}_9trait.tsv'
        raw={}
    if snps is None:raise RuntimeError(f'Missing input: {trait}')
    registry=pd.read_csv(MAP/'gene_registry.tsv',sep='\t')
    parts={}; totals={}
    for chrom in range(1,23):
        prefix=f'chr{chrom:02d}'
        gi=np.load(MAP/f'{prefix}_gene_index.npy',mmap_mode='r')
        vi=np.load(MAP/f'{prefix}_variant_index.npy',mmap_mode='r')
        pos=np.load(MAP/f'{prefix}_positions.npy',mmap_mode='r')
        pv=np.array([snps.get((chrom,int(p)),{}).get('p',np.nan) for p in pos])
        edges=np.r_[0,np.flatnonzero(np.diff(gi)!=0)+1,len(gi)]
        for a,b in zip(edges[:-1],edges[1:]):
            idx=int(gi[a]); p=pv[vi[a:b]]; p=p[np.isfinite(p)]
            parts.setdefault(idx,[]).append(p); totals[idx]=totals.get(idx,0)+b-a
    del snps
    rows=[]
    for idx in range(len(registry)):
        p=np.concatenate(parts[idx]); gene=registry.iloc[idx].gene
        legacy=old.acat(p); stable=stable_acat(p)
        n1=int((p==1).sum())
        rows.append(dict(gene=gene,trait=trait,nsnps_matched=len(p),nsnps_total=totals[idx],replay_p=legacy,stable_p=stable,min_p=float(p.min()) if len(p) else np.nan,n_input_one=n1,n_input_at_floor=int((p==1e-300).sum()),n_input_below_1e_16=int((p<1e-16).sum()),mathematical_P1_boundary_p=1. if n1 else stable))
    result=pd.DataFrame(rows)
    original=pd.read_csv(saved,sep='\t',float_precision='round_trip')
    result=result.merge(original[['gene','acat_p','nsnps_matched','min_p']],on='gene',how='left',validate='one_to_one',suffixes=('','_source'))
    valid=result.replay_p.notna()
    tests=[dict(check='evaluable_membership',passed=bool(valid.equals(result.acat_p.notna()))),dict(check='matched_count',passed=bool(np.array_equal(result.loc[valid,'nsnps_matched'],result.loc[valid,'nsnps_matched_source']))),dict(check='legacy_p_replay',passed=bool(np.allclose(result.loc[valid,'replay_p'],result.loc[valid,'acat_p'],rtol=1e-12,atol=0))),dict(check='minimum_p',passed=bool(np.allclose(result.loc[valid,'min_p'],result.loc[valid,'min_p_source'],rtol=1e-12,atol=0))),dict(check='mapping_counts',passed=bool(np.array_equal(result.nsnps_total,registry.nsnps_cis)))]
    write(result,f'{trait}_gene_acat.tsv'); write(pd.DataFrame(tests),f'{trait}_replay_checks.tsv')
    if not all(t['passed'] for t in tests):raise AssertionError((trait,tests))
    assert sha(path)==before
    manifest=dict(trait=trait,started_utc=start,completed_utc=datetime.now(timezone.utc).isoformat(),input_path=str(path.relative_to(ROOT)),input_sha256=before,saved_gene_source=str(saved.relative_to(ROOT)),saved_gene_sha256=sha(saved),script_sha256=sha(Path(__file__)),n_genes=len(result),n_evaluable=int(valid.sum()),legacy_replay_passed=True,raw_adapter_audit=raw)
    (OUT/f'{trait}_manifest.json').write_text(json.dumps(manifest,indent=2,default=str))
    print(f'{trait}: complete, replay PASS, {int(valid.sum())} evaluable',flush=True)

if __name__=='__main__':
    parser=argparse.ArgumentParser();parser.add_argument('traits',nargs='+',choices=TRAITS);args=parser.parse_args()
    for trait in args.traits:run(trait)
